select outlet_size,
cast(sum(total_sales) as decimal(10, 2)) as total_sales,
cast((sum(total_sales) * 100.0 / sum(sum(total_sales))
over()) as decimal(10, 2)) as sales_percentage
from grocery_data
group by outlet_size order by total_sales desc;


query explanation
(sum(total_sales) = sum of total sales
sum(sum(total_sales)) over()) = grand total of sales for all outlet sizes
casting the entire thing to decimal with 2 decimal places
multiplying the result with 100
